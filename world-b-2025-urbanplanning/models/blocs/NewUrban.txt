species urbanplanning parent: bloc {
	string name <- "urbanplanning";
	urban_producer producer <- nil;
	urban_consumer consumer <- nil;
	map<string, int> to_build <- [];
	list<mini_city> mini_cities <- [];

	action setup {
		list<urban_producer> producers <- [];
		list<urban_consumer> consumers <- [];
		
		int total_mini_cities <- length(mini_cities);
		
		if total_mini_cities < 3 {
			write "ERROR: Not enough mini_cities (" + total_mini_cities + "). Minimum: 3";
			return;
		}
		
		loop c over: mini_cities {do initialize_city_buildings(c);}
		create urban_producer number: 1 returns: producers;
		create urban_consumer number: 1 returns: consumers;
		producer <- first(producers);
		consumer <- first(consumers);
	}
	
	action tick (list<human> pop) {
		
		if (cycle > 0){
			write "CYCLE "+ int(cycle);
			loop o over: production_outputs_U{
				write o + ": " + tick_production_U[o];
			}
		}
		
		ask residents {
			myself.mini_cities <- self.mini_cities;
		}

		if (length(mini_cities) < 1){
			write "*** ERROR: List of mini-cities obtained in the tick of an empty urbanplanning: " + mini_cities;
		}
		
		do collect_last_tick_data();
		do population_activity(pop);
		do check_building_queue();
		do calculate_minicities_demand();
		do do_minicities_production();
	}

	production_agent get_producer {
		return producer;
	}

	list<string> get_output_resources_labels {
		return production_outputs_U + production_inner_material_U;
	}

	list<string> get_input_resources_labels {
		return production_inputs_U;
	}

	list<string> get_autoproduction_resources_labels {
		return production_inner_material_U;
	}

	action set_external_producer (string product, bloc bloc_agent) {
	}

	action collect_last_tick_data {
		if (cycle > 0) { 
			tick_consumption_U <- consumer.get_tick_consumption(); 
			tick_resources_used_U <- producer.get_tick_inputs_used(); 
			tick_production_U <- producer.get_tick_outputs_produced(); 
			tick_emissions_U <- producer.get_tick_emissions(); 
			
			ask urban_consumer { 
				do reset_tick_counters;
			}
			ask urban_producer { 
				do reset_tick_counters;
			}
		}
	}

	action do_minicities_production {
		loop mini_ville over: mini_cities {
			ask urban_producer {
				do produce_city(mini_ville);
			}
		}
	}

	action calculate_minicities_demand {
		loop c over: mini_cities {
			int current_wooden_pop <- int(c.pop * wooden_building_ratio);
			int current_modular_pop <- c.pop - current_wooden_pop;
			c.demand["wooden house"] <- float(ceil(current_wooden_pop / max_capacity["wooden house"]));
			float needed_lobbies <- float(ceil(current_modular_pop / max_capacity["modular house"]));
			c.demand["modular house"] <- needed_lobbies;
			float base_capacity_used <- needed_lobbies * min_modular_capacity;
			c.demand["modular house extension"] <- max([0.0, current_modular_pop - base_capacity_used]);
			c.demand["school"] <- float(ceil(c.go_to_school / max_capacity["school"]));
			c.demand["company"] <- float(ceil(c.go_to_work * company_per_worker / max_capacity["company"]));
			c.demand["plastic factory"] <- float(ceil(c.go_to_work * factory_per_worker / max_capacity["plastic factory"]));
			int nb_free <- int(ceil(c.go_to_work * unemployed_per_worker)) + (c.pop - c.go_to_work - c.go_to_school);
			c.demand["leisure"] <- float(ceil(nb_free / max_capacity["leisure"]));
			loop k over: c.demand.keys {
            	c.shortage[k] <- max(0.0, c.demand[k] - c.building_supply[k]);
			}	
		}
	}

	action check_building_queue {
		loop mc over: mini_cities {
			list<building_project> tmp <- [];
			loop project over: mc.queue {
				project.completion_time <- project.completion_time - 1;
				if (project.completion_time <= 0){
					string building_name <- project.building;
					int building_quantity <- int(project.quantity);
					mc.building_supply[building_name] <- mc.building_supply[building_name] + building_quantity;
					mc.tick_production[building_name] <- mc.tick_production[building_name] + building_quantity;
					ask producer{
						self.tick_production[building_name] <- self.tick_production[building_name] + building_quantity;
					}
					if (building_quantity > 0){
						if (building_name = "wooden house") {
							create wooden_house number: building_quantity { my_city <- mc; }
						} else if (building_name = "modular house") {
							create modular_house number: building_quantity { my_city <- mc; }
						} else if (building_name = "plastic factory") {
							create factory number: building_quantity { my_city <- mc; }
						} else if (building_name = "company") {
							create company number: building_quantity { my_city <- mc; }
						} else if (building_name = "school") {
							create school number: building_quantity { my_city <- mc; }
						} else if (building_name = "leisure") {
							create leisure number: building_quantity { my_city <- mc; }
						}
					}
					tmp <- tmp + [project];
				} 
			}
			mc.queue <- mc.queue - tmp;
			ask tmp{
				do die;
			}
		}
	}
	
	action population_activity (list<human> pop) {
		ask pop { 
			ask myself.urban_consumer {
				do consume(myself);
			}
		}
	}

	species urban_producer parent: production_agent {
		map<string, bloc> external_producers;
		map<string, float> tick_resources_used <- [];
		map<string, float> tick_production <- [];
		map<string, float> tick_emissions <- [];
		map<string, float> tick_demand <- [];

		init {
			external_producers <- []; 
		}

		map<string, float> get_tick_inputs_used { return tick_resources_used; }
		map<string, float> get_tick_outputs_produced { return tick_production; }
		map<string, float> get_tick_emissions { return tick_emissions; }
		map<string, float> get_tick_demand { return tick_demand; }

		action reset_tick_counters { 
			loop a over: production_inner_material_U {
				tick_resources_used[a] <- 0.0; 
				tick_production[a] <- 0.0;
			}
			loop u over: production_inputs_U {
				tick_resources_used[u] <- 0.0; 
			}
			loop p over: production_outputs_U {
				tick_production[p] <- 0.0; 
				tick_demand[p] <- 0.0;
			}
			loop e over: production_emissions_U {
				tick_emissions[e] <- 0.0;
			}
		}
		
		bool produce_city (mini_city mini_ville) {
			bool ok <- true;

			float max_plastic_capacity <- mini_ville.building_supply["plastic factory"] * factory_productivity;
			float recycled_plastic <- mini_ville.available_budget["kg plastic"];
			
			float total_plastic_needed <- 0.0;
			loop c over: mini_ville.shortage.keys {
				if (mini_ville.shortage[c] > 0 and production_output_inputs_U[c].keys contains "kg plastic") {
					total_plastic_needed <- total_plastic_needed + (mini_ville.shortage[c] * production_output_inputs_U[c]["kg plastic"]);
				}
			}

			float plastic_to_produce <- max([0.0, total_plastic_needed - recycled_plastic]);
			float actual_plastic_produced <- min([plastic_to_produce, max_plastic_capacity]);
			float total_available_plastic <- recycled_plastic + actual_plastic_produced;

			map<string, float> projects_to_build <- [];
			map<string, float> needed_external <- [];
			map<string, float> planned_budget_usage <- [];
			float remaining_plastic <- total_available_plastic;

			if (actual_plastic_produced > 0) {
				loop u over: production_output_inputs_U["kg plastic"].keys {
					float req <- actual_plastic_produced * production_output_inputs_U["kg plastic"][u];
					if !(needed_external.keys contains u) { needed_external[u] <- 0.0; }
					needed_external[u] <- needed_external[u] + req;
				}
			}

			loop c over: mini_ville.shortage.keys {
				float qty <- mini_ville.shortage[c];
				if (qty > 0) {
					if (production_output_inputs_U[c].keys contains "kg plastic") {
						float plastic_per_unit <- production_output_inputs_U[c]["kg plastic"];
						if (plastic_per_unit > 0) {
							float max_by_plastic <- float(remaining_plastic / plastic_per_unit);
							qty <- min([qty, max_by_plastic]);
							remaining_plastic <- remaining_plastic - (qty * plastic_per_unit);
						}
					}

					if (qty > 0) {
						projects_to_build[c] <- qty;
						
						loop u over: production_inputs_U {
							if (production_output_inputs_U[c].keys contains u and u != "kg plastic") {
								float total_req <- qty * production_output_inputs_U[c][u];

								if (mini_ville.available_budget.keys contains u) {
									float used_recycled <- 0.0;
									if (planned_budget_usage.keys contains u) {
										float left_budget <- max([0.0, mini_ville.available_budget[u] - planned_budget_usage[u]]);
										used_recycled <- min([total_req, left_budget]);
										planned_budget_usage[u] <- planned_budget_usage[u] + used_recycled;
									} else {
										used_recycled <- min([total_req, mini_ville.available_budget[u]]);
										planned_budget_usage[u] <- used_recycled;
									}
									total_req <- total_req - used_recycled;
								}

								if (total_req > 0) {
									if !(needed_external.keys contains u) { needed_external[u] <- 0.0; }
									needed_external[u] <- needed_external[u] + total_req;
								}
							}
						}
					}
				}
			}

			if (empty(projects_to_build) and actual_plastic_produced = 0.0) { return true; }

			bool can_build_everything <- true;
			loop u over: needed_external.keys {
				if (external_producers.keys contains u) {
					bool av <- external_producers[u].producer.produce([u::needed_external[u]]);
					if not av {
						can_build_everything <- false;
						break;
					}
				}
			}

			if (can_build_everything) {
				planned_budget_usage["kg plastic"] <- total_available_plastic - remaining_plastic - actual_plastic_produced;
				if (planned_budget_usage["kg plastic"] > 0) {
					mini_ville.available_budget["kg plastic"] <- max([0.0, mini_ville.available_budget["kg plastic"] - planned_budget_usage["kg plastic"]]);
				}
				loop u over: planned_budget_usage.keys {
					if (u != "kg plastic") {
						mini_ville.available_budget[u] <- max([0.0, mini_ville.available_budget[u] - planned_budget_usage[u]]);
					}
				}

				loop u over: needed_external.keys {
					tick_resources_used[u] <- tick_resources_used[u] + needed_external[u];
					mini_ville.tick_resources_used[u] <- mini_ville.tick_resources_used[u] + needed_external[u];
				}

				if (actual_plastic_produced > 0) {
					tick_production["kg plastic"] <- tick_production["kg plastic"] + actual_plastic_produced;
					mini_ville.tick_production["kg plastic"] <- mini_ville.tick_production["kg plastic"] + actual_plastic_produced;

					if (production_output_emissions_U.keys contains "kg plastic") {
						loop e over: production_emissions_U {
							if (production_output_emissions_U["kg plastic"].keys contains e) {
								tick_emissions[e] <- tick_emissions[e] + (actual_plastic_produced * production_output_emissions_U["kg plastic"][e]);
							}
						}
					}
				}

				loop c over: projects_to_build.keys {
					float qty <- projects_to_build[c];

					if (production_output_emissions_U.keys contains c) {
						loop e over: production_emissions_U {
							if (production_output_emissions_U[c].keys contains e) {
								tick_emissions[e] <- tick_emissions[e] + (qty * production_output_emissions_U[c][e]);
							}
						}
					}

					string queue_key <- c + "_" + time_cost_U[c];
					mini_ville.building_queue[queue_key] <- [c, qty, time_cost_U[c]];
					mini_ville.potential_building_supply[c] <- mini_ville.potential_building_supply[c] + qty;

					ask mini_ville {
						create building_project {
							building <- c;
							quantity <- qty;
							completion_time <- time_cost_U[c];
							myself.queue <- myself.queue + self;
						}
					}
				}
			} else {
				ok <- false;
			}

			return ok;
		}

		bool produce (map<string, float> demand) {

	species house{
		string house_type;
		int age <- 0;
		int lifespan;
		int capacity;
		float proba_renew update: renew_proba[house_type];
		mini_city my_city;
		
		reflex renewal{
			age <- age + 1;
			bool renew <- flip(proba_renew);
			if (age >= lifespan or renew) {
				ask my_city{
					building_supply[myself.house_type] <- max([0.0, building_supply[myself.house_type] - 1]);
					loop k over: recycled_ratio.keys{
						if (production_output_inputs_U[myself.house_type].keys contains k) {
							available_budget[k] <- available_budget[k] + (recycled_ratio[k] * production_output_inputs_U[myself.house_type][k]);
						}
					}
				}
				do die;
			}
		}
	}

	action initialize_city_buildings(mini_city c){
		int target_wooden_capacity <- int(c.pop * wooden_building_ratio);
    	int target_modular_capacity <- c.pop - target_wooden_capacity;
    	
    	int nb_wooden <- int(ceil(target_wooden_capacity / max_capacity["wooden house"]));
    	int nb_modular <- int(ceil(target_modular_capacity / min_modular_capacity));
    	
    	create wooden_house number: nb_wooden{
    		my_city <- c;
    		age <- rnd(0, lifespan - 1);
    	}
    	create modular_house number: nb_modular{
    		my_city <- c;
    		age <- rnd(0, lifespan - 1);
    	}
    	
    	int nb_school <- int(ceil(c.go_to_school / max_capacity["school"]));
    	create school number: nb_school{
    		my_city <- c;
    		age <- rnd(0, lifespan - 1);
    	}
    	
    	int nb_company <- int(ceil(c.go_to_work * company_per_worker / max_capacity["company"]));
    	create company number: nb_company{
    		my_city <- c;
    		age <- rnd(0, lifespan - 1);
    	}
    	
    	int nb_factory <- int(ceil(c.go_to_work * factory_per_worker / max_capacity["plastic factory"]));
    	create factory number: nb_factory{
    		my_city <- c;
    		age <- rnd(0, lifespan - 1);
    	}
    	
    	int nb_free <- int(ceil(c.go_to_work * unemployed_per_worker)) + (c.pop - c.go_to_work - c.go_to_school);
    	int nb_leisure <- int(ceil(nb_free / max_capacity["leisure"]));
    	create leisure number: nb_leisure{
    		my_city <- c;
    		age <- rnd(0, lifespan - 1);
    	}
    	
    	c.building_supply["modular house"] <- float(nb_modular);
    	c.building_supply["wooden house"] <- float(nb_wooden);
    	c.building_supply["plastic factory"] <- float(nb_factory);
    	c.building_supply["company"] <- float(nb_company);
